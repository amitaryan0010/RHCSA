awk — Text Processing & Pattern Scanning
    awk processes text field by field (columns). It's ideal for structured data.

Basic Syntax: # awk 'pattern { action }' filename

$0 = Entire line
$1 = First field (column)
$2 = Second field
$NF = Last field
NR = Line number
NF = Number of fields in current line

#######################################################################################
Example 

1: Print specific columns
    # ps -elf | awk '{print $1, $3, $8}'

2: Print with header condition
    # ps -elf | awk 'NR==1 || /ssh/'        --> Prints header + all lines matching "ssh".

3: Print only processes with nice value != 0
    # ps -elf | awk '$8 != 0'               --> Prints only processes where NI column is not zero.

4: Column arithmetic
    File: scores.txt
        Alice 80 90 85
        Bob 70 75 80
        Charlie 90 95 92

    # awk '{sum = $2 + $3 + $4; print $1, sum, sum/3}' scores.txt
    Output
        Alice 255 85
        Bob 225 75
        Charlie 277 92.3333

5: Filter with conditions
    # awk '$2 > 80 {print $1, "Excellent"}' scores.txt
    Output:
        Alice Excellent
        Charlie Excellent

6: Field separator change
    File: data.csv
        name,age,city
        Alice,30,NY
        Bob,25,LA

    #awk -F',' '{print $1, $3}' data.csv            --> -F',' = Sets field separator to comma
    Output:
        name city
        Alice NY
        Bob LA

REAL TIME EXAMPLE, extact the IP from # ip a command

[root@rhel-9 ~]# ip a
1: lo: <LOOPBACK,UP,LOWER_UP> mtu 65536 qdisc noqueue state UNKNOWN group default qlen 1000
    link/loopback 00:00:00:00:00:00 brd 00:00:00:00:00:00
    inet 127.0.0.1/8 scope host lo
       valid_lft forever preferred_lft forever
    inet6 ::1/128 scope host
       valid_lft forever preferred_lft forever
2: eth0: <BROADCAST,MULTICAST,UP,LOWER_UP> mtu 1500 qdisc fq_codel state UP group default qlen 1000
    link/ether 08:00:27:6e:11:22 brd ff:ff:ff:ff:ff:ff
    altname enp0s3
    inet 192.168.0.128/24 brd 192.168.0.255 scope global dynamic noprefixroute eth0
       valid_lft 79390sec preferred_lft 79390sec
3: eth1: <BROADCAST,MULTICAST,UP,LOWER_UP> mtu 1500 qdisc fq_codel state UP group default qlen 1000
    link/ether 08:00:27:f6:2e:3b brd ff:ff:ff:ff:ff:ff
    altname enp0s8
    inet 192.168.139.5/24 brd 192.168.139.255 scope global dynamic noprefixroute eth1
       valid_lft 433sec preferred_lft 433sec


** We can see IP starts with inet so grep this keyword:
[root@rhel-9 ~]# ip a | grep -i net
    inet 127.0.0.1/8 scope host lo
    inet6 ::1/128 scope host
    inet 192.168.0.128/24 brd 192.168.0.255 scope global dynamic noprefixroute eth0
    inet 192.168.139.5/24 brd 192.168.139.255 scope global dynamic noprefixroute eth1

** now, we can see IP is in second column, so get it using awk command
[root@rhel-9 ~]# ip a | grep -i net | awk '{print $2}'
127.0.0.1/8
::1/128
192.168.0.128/24
192.168.139.5/24

** now, get rid of separator here which is forward slash (/)
[root@rhel-9 ~]# ip a | grep -i net | awk '{print $2}' | awk -F '/' '{print $1}'
127.0.0.1
::1
192.168.0.128
192.168.139.5

** We have the final IP

** If we want to replace this . with :, then make use of tr command

[root@rhel-9 ~]# ip a | grep -i net | awk '{print $2}' | awk -F '/' '{print $1}' | tr '.' ':'
127:0:0:1
::1
192:168:0:128
192:168:139:5
